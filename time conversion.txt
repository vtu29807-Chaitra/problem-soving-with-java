class Result {
     public static String timeConversion(String s) {
    
        // Extract AM or PM indicator
        String period = s.substring(8);
        
        // Extract numerical hour component
        int hour = Integer.parseInt(s.substring(0, 2));
        
        // Extract middle section (:MM:SS)
        String middle = s.substring(2, 8);
        
        if (period.equals("AM")) {
            if (hour == 12) {
                hour = 0; // 12:XX:XXAM -> 00:XX:XX
            }
        } else { // PM case
            if (hour != 12) {
                hour += 12; // 01:XX:XXPM -> 13:XX:XX (12 PM stays 12)
            }
        }
        
        // Format hour with leading zero if needed
        return String.format("%02d", hour) + middle;
    }

}
    

public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));
        BufferedWriter bufferedWriter = new BufferedWriter(new FileWriter(System.getenv("OUTPUT_PATH")));

        String s = bufferedReader.readLine();

        String result = Result.timeConversion(s);

        bufferedWriter.write(result);
        bufferedWriter.newLine();

        bufferedReader.close();
        bufferedWriter.close();
    }
}
